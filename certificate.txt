###############################################################################
# JCEKS / DEMOTRUST CONFIGURATION
###############################################################################

JCEKS_FILE=""
DEMO_TRUST_FILE=""

if [ -f "$KEYSTORE_INPUT_FILE" ]
then
    while IFS='=' read -r KEY VALUE
    do
        # Skip blank lines
        [ -z "$KEY" ] && continue

        # Skip comments
        [[ "$KEY" == \#* ]] && continue

        # Remove leading/trailing spaces
        KEY=$(echo "$KEY" | xargs)
        VALUE=$(echo "$VALUE" | xargs)

        case "$KEY" in

            JCEKS_FILE)
                JCEKS_FILE="$VALUE"
                ;;

            DEMO_TRUST_FILE)
                DEMO_TRUST_FILE="$VALUE"
                ;;

        esac

    done < "$KEYSTORE_INPUT_FILE"
fi

---------------------------------------------------------------------------------------------------------------------------------------------------------------------
###############################################################################
# JCEKS CERTIFICATE
###############################################################################

echo "================================================================"
echo "                    JCEKS CERTIFICATE CHECK"
echo "================================================================"
echo ""

if [ -n "$JCEKS_FILE" ]
then

    echo "------------------------------------------------------------"
    echo "Checking JCEKS : $JCEKS_FILE"
    echo "Alias          : $JCEKS_ALIAS"
    echo "------------------------------------------------------------"

    if [ ! -f "$JCEKS_FILE" ]
    then

        echo "Status : JCEKS FILE NOT FOUND"
        echo ""

        write_alert \
            "JCEKS Certificate" \
            "File : $JCEKS_FILE | Alias : $JCEKS_ALIAS" \
            "N/A" \
            "N/A" \
            "JCEKS FILE NOT FOUND"

    else

        JCEKS_OUTPUT=$(
            keytool -list -v \
                -keystore "$JCEKS_FILE" \
                -storetype JCEKS \
                -alias "$JCEKS_ALIAS" \
                2>/dev/null
        )

        if [ -z "$JCEKS_OUTPUT" ]
        then

            echo "Status : CERTIFICATE NOT FOUND / UNABLE TO READ"
            echo ""

            write_alert \
                "JCEKS Certificate" \
                "File : $JCEKS_FILE | Alias : $JCEKS_ALIAS" \
                "N/A" \
                "N/A" \
                "CERTIFICATE NOT FOUND"

        else

            ###################################################################
            # Extract certificate expiry date
            #
            # Example keytool output:
            #
            # Valid from: Mon Aug 10 10:00:00 IST 2025
            # until: Thu Aug 10 10:00:00 IST 2026
            ###################################################################

            JCEKS_END_DATE=$(
                echo "$JCEKS_OUTPUT" |
                grep "Valid from:" |
                sed 's/.*until: //'
            )

            if [ -z "$JCEKS_END_DATE" ]
            then

                echo "Status : UNABLE TO DETERMINE CERTIFICATE EXPIRY"
                echo ""

                write_alert \
                    "JCEKS Certificate" \
                    "File : $JCEKS_FILE | Alias : $JCEKS_ALIAS" \
                    "N/A" \
                    "N/A" \
                    "UNABLE TO DETERMINE EXPIRY"

            else

                ################################################################
                # Convert expiry date to epoch
                ################################################################

                JCEKS_END_EPOCH=$(date -d "$JCEKS_END_DATE" +%s 2>/dev/null)

                if [ -z "$JCEKS_END_EPOCH" ]
                then

                    echo "Expiry Date : $JCEKS_END_DATE"
                    echo "Status      : INVALID EXPIRY DATE"
                    echo ""

                    write_alert \
                        "JCEKS Certificate" \
                        "File : $JCEKS_FILE | Alias : $JCEKS_ALIAS" \
                        "$JCEKS_END_DATE" \
                        "N/A" \
                        "INVALID EXPIRY DATE"

                else

                    ################################################################
                    # Calculate days remaining
                    ################################################################

                    JCEKS_DAYS_LEFT=$(
                        ( JCEKS_END_EPOCH - NOW_EPOCH ) / 86400
                    )

                    ################################################################
                    # Determine status
                    ################################################################

                    if [ "$JCEKS_END_EPOCH" -le "$NOW_EPOCH" ]
                    then

                        JCEKS_STATUS="EXPIRED"

                    elif [ "$JCEKS_END_EPOCH" -le $((NOW_EPOCH + THRESHOLD_DAYS * 86400)) ]
                    then

                        JCEKS_STATUS="EXPIRING SOON"

                    else

                        JCEKS_STATUS="VALID"

                    fi

                    ################################################################
                    # Display result
                    ################################################################

                    echo "Certificate : $JCEKS_ALIAS"
                    echo "Expiry Date : $JCEKS_END_DATE"
                    echo "Days Left   : $JCEKS_DAYS_LEFT"
                    echo "Status      : $JCEKS_STATUS"
                    echo ""

                    ################################################################
                    # Alert if certificate is not valid
                    ################################################################

                    if [ "$JCEKS_STATUS" != "VALID" ]
                    then

                        write_alert \
                            "JCEKS Certificate" \
                            "File : $JCEKS_FILE | Alias : $JCEKS_ALIAS" \
                            "$JCEKS_END_DATE" \
                            "$JCEKS_DAYS_LEFT" \
                            "$JCEKS_STATUS"

                    fi

                fi

            fi

        fi

    fi

else

    echo "JCEKS_FILE is not configured."
    echo ""

fi
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
###############################################################################
# DEMOTRUST.JKS CERTIFICATES
###############################################################################

echo "================================================================"
echo "                 DEMOTRUST.JKS CERTIFICATE CHECK"
echo "================================================================"
echo ""

if [ -n "$DEMO_TRUST_FILE" ]
then

    echo "------------------------------------------------------------"
    echo "Checking TrustStore : $DEMO_TRUST_FILE"
    echo "------------------------------------------------------------"

    if [ ! -f "$DEMO_TRUST_FILE" ]
    then

        echo "Status : TRUSTSTORE FILE NOT FOUND"
        echo ""

        write_alert \
            "demoTrust.jks" \
            "File : $DEMO_TRUST_FILE" \
            "N/A" \
            "N/A" \
            "TRUSTSTORE FILE NOT FOUND"

    else

        ########################################################################
        # Read complete truststore
        #
        # We intentionally use keytool -list -v.
        # No certificate export is required.
        ########################################################################

        TRUSTSTORE_OUTPUT=$(
            keytool -list -v \
                -keystore "$DEMO_TRUST_FILE" \
                -storetype JKS \
                2>/dev/null
        )

        if [ -z "$TRUSTSTORE_OUTPUT" ]
        then

            echo "Status : UNABLE TO READ TRUSTSTORE"
            echo ""

            write_alert \
                "demoTrust.jks" \
                "File : $DEMO_TRUST_FILE" \
                "N/A" \
                "N/A" \
                "UNABLE TO READ TRUSTSTORE"

        else

            ####################################################################
            # Process every certificate entry.
            #
            # keytool -list -v gives blocks similar to:
            #
            # Alias name: example
            # Creation date: ...
            # Entry type: trustedCertEntry
            #
            # Owner: ...
            # Valid from: ... until: ...
            ####################################################################

            echo "$TRUSTSTORE_OUTPUT" |
            awk '
            BEGIN {
                alias=""
                expiry=""
            }

            /^Alias name:/ {
                alias=$0
                sub(/^Alias name: /, "", alias)
            }

            /Valid from:/ {

                expiry=$0
                sub(/^.*until: /, "", expiry)

                if (alias != "") {
                    print alias "|" expiry
                }

                alias=""
                expiry=""
            }

            ' |
            while IFS='|' read -r TRUST_ALIAS TRUST_END_DATE
            do

                ################################################################
                # Skip invalid entries
                ################################################################

                [ -z "$TRUST_ALIAS" ] && continue
                [ -z "$TRUST_END_DATE" ] && continue

                echo ""
                echo "------------------------------------------------------------"
                echo "Certificate : $TRUST_ALIAS"
                echo "------------------------------------------------------------"

                ################################################################
                # Convert expiry date to epoch
                ################################################################

                TRUST_END_EPOCH=$(date -d "$TRUST_END_DATE" +%s 2>/dev/null)

                if [ -z "$TRUST_END_EPOCH" ]
                then

                    echo "Expiry Date : $TRUST_END_DATE"
                    echo "Status      : INVALID EXPIRY DATE"
                    echo ""

                    write_alert \
                        "demoTrust.jks Certificate" \
                        "File : $DEMO_TRUST_FILE | Alias : $TRUST_ALIAS" \
                        "$TRUST_END_DATE" \
                        "N/A" \
                        "INVALID EXPIRY DATE"

                    continue

                fi

                ################################################################
                # Calculate days remaining
                ################################################################

                TRUST_DAYS_LEFT=$(
                    ( TRUST_END_EPOCH - NOW_EPOCH ) / 86400
                )

                ################################################################
                # Determine status
                ################################################################

                if [ "$TRUST_END_EPOCH" -le "$NOW_EPOCH" ]
                then

                    TRUST_STATUS="EXPIRED"

                elif [ "$TRUST_END_EPOCH" -le $((NOW_EPOCH + THRESHOLD_DAYS * 86400)) ]
                then

                    TRUST_STATUS="EXPIRING SOON"

                else

                    TRUST_STATUS="VALID"

                fi

                ################################################################
                # Display result
                ################################################################

                echo "Expiry Date : $TRUST_END_DATE"
                echo "Days Left   : $TRUST_DAYS_LEFT"
                echo "Status      : $TRUST_STATUS"
                echo ""

                ################################################################
                # Alert if not valid
                ################################################################

                if [ "$TRUST_STATUS" != "VALID" ]
                then

                    write_alert \
                        "demoTrust.jks Certificate" \
                        "File : $DEMO_TRUST_FILE | Alias : $TRUST_ALIAS" \
                        "$TRUST_END_DATE" \
                        "$TRUST_DAYS_LEFT" \
                        "$TRUST_STATUS"

                fi

            done

        fi

    fi

else

    echo "DEMO_TRUST_FILE is not configured."
    echo ""

fi
